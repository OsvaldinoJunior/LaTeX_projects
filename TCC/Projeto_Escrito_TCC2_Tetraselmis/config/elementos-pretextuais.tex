% Elementos iniciais com a fonte, as margens e os títulos do estilo do trabalho.
\newcounter{apendice}
\renewcommand{\theapendice}{\Alph{apendice}}
\newcommand{\elementopretextual}[2]{%
  \clearpage
  \pdfbookmark[1]{#1}{#2}%
  {\centering\bfseries\MakeUppercase{#1}\par}%
  \vspace{\baselineskip}%
}
\newcommand{\listapretextual}[3]{%
  \elementopretextual{#1}{#2}%
  \begingroup
  \makeatletter\csname @starttoc\endcsname{#3}\makeatother
  \endgroup
}
