% SLIDE 2 — DEFINIÇÃO | 1:00
\begin{frame}[t]{Uvas brancas, outra forma de vinificar}
  \framesubtitle{A diferença central é o contato do mosto com as partes sólidas da uva}
  \vspace{0.20cm}
  \bodytext{Vinho laranja é um vinho de \textbf{uvas brancas fermentado com as cascas},
    também chamado de vinho âmbar ou branco de maceração.}
  \vspace{0.27cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \card{BRANCO CONVENCIONAL}{Separar antes}{Em geral, as uvas são prensadas e o mosto
        fermenta separado das cascas.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \card{VINHO LARANJA}{Manter o contato}{Mosto e cascas permanecem juntos durante
        a fermentação; a maceração pode continuar depois.}
    \end{column}
  \end{columns}
  \callout{\textbf{O nome vem da cor:} é vinho de uvas, não da fruta laranja.}
  \source{[1], [4]}
\end{frame}

% SLIDE 4 — FLUXOGRAMA | 1:20
\begin{frame}[t]{Como o vinho laranja é produzido}
  \framesubtitle{Fluxo simplificado: fermentação e extração acontecem em paralelo}
  \vspace{0.30cm}
  \begin{tikzpicture}[
    stagebox/.style={draw=Honey,fill=Cream,rounded corners=2pt,text=Ink,
      align=center,text width=2.25cm,minimum height=1.85cm,inner sep=0.08cm,
      font=\fontsize{9.5}{12}\selectfont},
    flow/.style={-{Stealth[length=2mm]},Amber,line width=0.9pt}]
    \node[stagebox] (a) {\textbf{1}\\\textbf{Uvas brancas}\\Seleção e recepção};
    \node[stagebox,right=0.38cm of a] (b) {\textbf{2}\\\textbf{Preparo}\\Preparar o mosto};
    \node[stagebox,fill=Amber,text=white,right=0.38cm of b] (c) {\textbf{3}\\\textbf{Fermentação}\\Maceração com as cascas};
    \node[stagebox,right=0.38cm of c] (d) {\textbf{4}\\\textbf{Prensagem}\\Separação dos sólidos};
    \node[stagebox,right=0.38cm of d] (e) {\textbf{5}\\\textbf{Finalização}\\Maturação e envase};
    \draw[flow] (a) -- (b); \draw[flow] (b) -- (c);
    \draw[flow] (c) -- (d); \draw[flow] (d) -- (e);
  \end{tikzpicture}
  \vspace{0.30cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \kicker{FERMENTAÇÃO}
      \smalltext{Leveduras transformam açúcares em etanol e CO$_2$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \kicker{MACERAÇÃO}
      \smalltext{Extrai compostos das cascas e sementes; pode continuar após fermentar.}
    \end{column}
  \end{columns}
  \callout{\textbf{Categoria da OIV:} mínimo de \textbf{1 mês} de maceração. No mercado,
    o nome ``laranja'' também abrange contatos mais curtos.}

\end{frame}

% SLIDE 6 — PERFIL SENSORIAL | 1:00
\begin{frame}[t]{O que pode aparecer na taça?}
  \framesubtitle{Há uma família de estilos, e não um único perfil obrigatório}
  \vspace{0.30cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.315\textwidth}
      \card{VISUAL}{Dourado a âmbar}{A intensidade depende da uva, da maceração
        e da evolução do vinho.}
    \end{column}
    \begin{column}{0.315\textwidth}
      \card{AROMAS}{Diversidade}{Podem surgir frutas secas, casca cítrica e nozes,
        conforme o estilo.}
    \end{column}
    \begin{column}{0.315\textwidth}
      \card{BOCA}{Textura}{Pode haver mais estrutura e adstringência
        do que em brancos convencionais.}
    \end{column}
  \end{columns}
  \vspace{0.25cm}
  \smalltext{\textbf{Cor não determina doçura:} o açúcar residual depende
    da condução da fermentação.}
  \vspace{0.17cm}
  \smalltext{\textbf{Estilo não é uma receita única:} não exige ânfora
    nem ausência de aditivos.}
  \source{[4], [6]}
\end{frame}

% SLIDE 7 — CONTROLE DE PROCESSO | 1:00
\begin{frame}[t]{Como equilibrar extração e qualidade?}
  \framesubtitle{As escolhas de processo precisam acompanhar o perfil desejado}
  \vspace{0.23cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \kicker{01 \quad UVA E TEMPO DE CONTATO}
      \smalltext{Cultivar, maturação e duração da maceração influenciam
        a composição e a intensidade da extração.}
      \vspace{0.34cm}
      \kicker{03 \quad OXIGÊNIO}
      \smalltext{Seu manejo interfere na evolução da cor e dos aromas;
        exposição excessiva pode prejudicar o resultado.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \kicker{02 \quad FERMENTAÇÃO}
      \smalltext{Acompanhar temperatura, atividade das leveduras e consumo
        de açúcar ajuda a conduzir o processo.}
      \vspace{0.34cm}
      \kicker{04 \quad HIGIENE E ACOMPANHAMENTO}
      \smalltext{Controlar alterações microbianas e avaliar o vinho ao longo
        do contato com os sólidos.}
    \end{column}
  \end{columns}
  \vspace{0.25cm}
  \callout{\textbf{Mais extração não garante mais qualidade.} O objetivo é o equilíbrio.}
  \source{[4], [5]}
\end{frame}
