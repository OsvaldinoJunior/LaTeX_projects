\newcommand{\photo}[3]{%
  \sbox{\PhotoBox}{\includegraphics[height=#3]{#1}}%

    \sbox{\PhotoBox}{\includegraphics[width=#2]{#1}}%
  \fi
  \begin{tikzpicture}[x=1cm,y=1cm]
    \begin{pgfinterruptboundingbox}
      \clip (0,0) rectangle (#2,#3);
      \node[inner sep=0pt] at ({0.5*(#2)},{0.5*(#3)}) {\usebox{\PhotoBox}};
    \end{pgfinterruptboundingbox}
    \path[use as bounding box] (0,0) rectangle (#2,#3);
  \end{tikzpicture}%
}
